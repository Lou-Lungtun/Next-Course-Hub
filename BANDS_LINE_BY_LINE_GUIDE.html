เอกสารนี้อธิบายโค้ดจริงของหน้า /bands ตั้งแต่ Type, Data, Page, Component และ CSS พร้อมอธิบายว่าคำสั่งแต่ละบรรทัดทำหน้าที่อะไร

ส่วนที่ 1: src/types/band.ts

ไฟล์นี้เป็นกติกาของข้อมูล TypeScript ยังไม่สร้างสิ่งที่มองเห็นบนหน้าเว็บ แต่ช่วยตรวจว่าข้อมูลถูกต้องก่อนรัน

export type Member = {
  id: number;
  name: string;
  role: string;
};

export type Band = {
  id: number;
  name: string;
  genre: string;
  image?: string;
  members: Member[];
};

ส่วนที่ 3: src/app/bands/page.tsx

ไฟล์นี้เป็น Parent Component และเป็นหน้า route /bands หน้าที่หลักคือรับข้อมูลทั้งหมดแล้ววนสร้าง BandCard

import BandCard from "@/components/BandCard";
import { bands } from "@/data/bandsdata";

export default function BandsPage() {
  return (
    <main className="page bandsPage">
      <header className="bandsHeader">
        <p className="eyebrow">FAVORITE BANDS</p>
        <h1>วงดนตรีที่ชื่นชอบ</h1>
        <p className="pageDescription">คำอธิบายหน้า</p>
      </header>

      <section className="bandGrid" aria-label="รายการวงดนตรี">
        {bands.map((band) => (
          <BandCard key={band.id} band={band} />
        ))}
      </section>
    </main>
  );
}

ส่วนที่ 4: src/components/BandCard.tsx

ไฟล์นี้เป็น Child Component รับข้อมูลหนึ่งวงผ่าน Props แล้วสร้างการ์ดหนึ่งใบ

import Image from "next/image";
import type { Band } from "@/types/band";

type BandCardProps = { band: Band };

export default function BandCard({ band }: BandCardProps) {
  return (
    <article className="bandCard">
      {band.image && (
        <div className="bandImageWrapper">
          <Image src={band.image} alt={`รูปวง ${band.name}`} />
        </div>
      )}

      <div className="bandContent">
        <p className="bandGenre">{band.genre}</p>
        <h2>{band.name}</h2>
        <ul className="memberList">
          {band.members.map((member) => (
            <li key={member.id}>
              <span>{member.name}</span>
              <small>{member.role}</small>
            </li>
          ))}
        </ul>
      </div>
    </article>
  );
}
บรรทัดคำอธิบาย
1–2import Image จาก Next.js เพื่อปรับขนาดและโหลดรูปอย่างเหมาะสม
3–4import Type Band มาใช้ตรวจ Props ของ Component
6–10สร้าง Type BandCardProps และกำหนดว่า prop ชื่อ band ต้องตรงกับ Type Band
12–14สร้าง Component BandCard, { band } คือ destructuring และ : BandCardProps ใช้ตรวจ Props
15return ( เริ่ม JSX ของการ์ด
17article เป็น semantic element สำหรับเนื้อหาที่สมบูรณ์หนึ่งรายการ ในที่นี้คือหนึ่งวง
19band.image && ( เป็น conditional rendering ถ้ามี image จึงแสดงส่วนรูป ถ้าไม่มีจะข้าม
20bandImageWrapper เป็นกรอบควบคุมสัดส่วนและส่วนเกินของรูป
22<Image เริ่ม Next.js Image Component
23className="bandImage" เชื่อมรูปกับ CSS
24src={band.image} อ่าน path รูปจากข้อมูลวงปัจจุบัน
25alt={`รูปวง ${band.name}`} ใช้ template literal รวมข้อความกับชื่อวง เพื่ออธิบายรูป
26–27width และ height กำหนดขนาดพื้นฐานและช่วยป้องกัน layout shift
28sizes บอกขนาดรูปตาม viewport: มือถือเต็มจอ, tablet ครึ่งจอ, desktop หนึ่งในสาม
29–31ปิด Image, div รูป และเงื่อนไข image
34bandContent ครอบข้อความทั้งหมดในการ์ด
36{band.genre} อ่านและแสดงแนวเพลงจาก Object
38{band.name} อ่านและแสดงชื่อวงใน h2
40h3 เป็นหัวข้อย่อย “สมาชิก”
42ul เริ่ม unordered list สำหรับรายการสมาชิก
44band.members.map() วนสมาชิกทุกคนของวงปัจจุบัน
45li key={member.id} สร้างหนึ่งรายการต่อหนึ่งคนและกำหนด key
47span แสดงชื่อสมาชิก
48small แสดงหน้าที่ด้วยตัวอักษรขนาดเล็กกว่า
49–55ปิด li, map, ul, bandContent, article, return และฟังก์ชัน

ส่วนที่ 5: CSS ใน src/app/globals.css

Selector เลือก element จาก className ใน JSX ส่วน property ภายในปีกกากำหนดรูปแบบของ element นั้น

.bandsPage และ .bandsHeader

คำสั่ง CSSหน้าที่
.bandsPageเลือก main ที่มี class bandsPage
position: relativeกำหนดขอบเขตอ้างอิงให้ element ลูกที่อาจใช้ position absolute
max-width: 52remจำกัดความกว้าง header เพื่อไม่ให้ข้อความยาวเกินไป
margin-bottom: 36pxเว้นระหว่าง header กับ Grid 36 พิกเซล
.bandsHeader h1เลือกเฉพาะ h1 ที่อยู่ภายใน bandsHeader
margin-bottom: 18pxเว้นระหว่าง h1 กับคำอธิบาย
margin: 0ลบ margin เริ่มต้นของ pageDescription

.bandGrid

display: gridเปิดใช้งาน CSS Grid

align-items: stretchยืดการ์ดในแถวให้สูงเท่ากัน

.bandCard

คำสั่ง CSSหน้าที่
position: relativeเป็นขอบเขตให้ pseudo-element ::after
